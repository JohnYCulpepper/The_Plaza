fastapi                     # the web framework everything is built on
uvicorn                     # runs your FastAPI app locally
sqlalchemy                  # lets you write Python instead of raw SQL
psycopg2-binary             # connects SQLAlchemy to PostgreSQL specifically
alembic                     # handles database migrations (updating tables safely)
python-jose[cryptography]   # creates and reads JWT tokens
passlib[bcrypt]             # hashes passwords so they're never stored plain
pydantic[email]             # validates request data, checks email format
python-multipart            # lets your API accept file uploads (store logos, product images)
stripe                      # connects to Stripe for payments
pillow                      # resizes and processes product images before storing them
python-dotenv               # reads your .env file so secrets stay out of your code
